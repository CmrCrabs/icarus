(defun zf/display-dired-sidetree ()
  "Displays Dired side tree."
  (interactive)
  (let ((buf (dired-noselect default-directory)))
    (display-buffer-in-side-window
     buf `((side . left)
           (slot . 0)
           (window-width . 30)
           (window-parameters . ((no-other-window . t)
                                 (no-delete-other-windows . t)
                                 (mode-line-format . ("" "%b"))))))
    (with-current-buffer buf
      (setq-local zf/filter-centaur t)
      (centaur-tabs-local-mode -1)
      (dired-omit-mode 1)
      (dired-hide-details-mode 1))))

(defun zf/todo-journal-logbook ()
  "Copy completed TODOs to that day's logbook."
  (interactive)
  (let* ((input-date (org-read-date))
         (entries (org-ql-query
                    :select '(let* ((heading (org-get-heading t t t t))
                                    (state (org-get-todo-state))
                                    (raw-body (org-get-entry))
                                    (clean-subheadings (if (string-match "\n\\*+ " raw-body)
                                                           (substring raw-body 0 (match-beginning 0))
                                                         raw-body))
                                    (body (replace-regexp-in-string
                                           "^[ \t]*:\\(LOGBOOK\\|PROPERTIES\\):\\(?:\n\\|.\\)*?\\(?:\n[ \t]*:END:\\)\n?"
                                           ""
                                           clean-subheadings))
                                    (tags-str (if (org-get-tags) (mapconcat 'identity (org-get-tags) ":") ""))
                                    (breadcrumbs-str (mapconcat 'identity (org-get-outline-path) ">"))
                                    (filename (file-name-nondirectory (buffer-file-name))))
                               (concat "**** *" state "* "
                                       "[" filename "] "
                                       (if (string-empty-p breadcrumbs-str) "" breadcrumbs-str)
                                       " | " heading
                                       (if (string-empty-p tags-str) "" (concat " :" tags-str ":"))
                                       "\n" body))
                    :from (org-agenda-files)
                    :where `(and (done) (closed :on ,input-date)))))
    (if entries
        (insert (mapconcat 'identity entries "\n\n"))
      (message "No TODOs found on %s" input-date))))

(defun zf/graph-fa2-network ()
  "Generate and display a ForceAtlas2 graph."
  (interactive)
  (let* ((buf (get-buffer-create "*graph-view*"))
         (files (directory-files-recursively (concat zf/curr-academic-path "/..") "\\.md\\'"))
         (file-ids (mapcar (lambda (file)
                             (replace-regexp-in-string (regexp-quote zf/curr-academic-path) "" file t t))
                           files))

         (id-base-table (let ((table (make-hash-table :test 'equal)))
                          (dolist (id file-ids)
                            (puthash (file-name-base id) id table))
                          table))
         (nodes (mapcar (lambda (id)
                          (list :id id :label (file-name-base id) :colour "#fd971e" :radius 2.0))
                        file-ids))
         (edges (with-temp-buffer
                  (let (edges)
                    (dolist (file files)
                      (erase-buffer)
                      (insert-file-contents file)
                      (goto-char (point-min))
                      (while (re-search-forward "\\[\\[\\([^]|#]+\\)" nil t)
                        (let ((source (gethash (file-name-base file) id-base-table))
                              (dest (gethash (file-name-base (match-string 1)) id-base-table)))
                          (when (and source dest)
                            (push (cons source dest)
                                  edges)))))
                    edges))))

    (with-current-buffer buf
      (add-hook 'graph-fa2-node-clicked-functions #'zf/graph-fa2-open-note nil t))

    (pop-to-buffer buf)
    (when (fboundp 'evil-emacs-state)
      (evil-insert-state))
    (graph-fa2-start buf nodes edges :cache-dir (expand-file-name "~/.emacs.d/graph-fa2-cache"))))

** pomodoro timer
#+begin_src emacs-lisp :tangle ~/.emacs.d/init.el
(setq zf/pomo-number 0)
(setq org-show-notification-handler #'ignore)

(defun zf/start-pomo-work ()
  "Start pomodoro work timer."
  (interactive)
  (setq zf/pomo-number (+ zf/pomo-number 1))
  (let ((str (format "pomodoro %1$s: " zf/pomo-number)))
    (setq org-timer-format (concat str "%s  "))
    (message "%s" (concat str "started"))
    (org-timer-set-timer "0:25:00")
    (setq org-timer-done-hook nil)
    (add-hook 'org-timer-done-hook (lambda ()
                                     (message "%s" (concat str "ended"))
                                     (org-capture nil "p")
                                     (ding)))))


(defun zf/start-pomo-break ()
  "Start pomodoro break timer, cycling long breaks."
  (interactive)
  (let ((str (format "break %1$s: " zf/pomo-number)))
    (if (= (% zf/pomo-number 4) 0)
        (progn
          (setq org-timer-format (concat "long " str "%s  "))
          (org-timer-set-timer "0:10:00"))
      (progn
        (setq org-timer-format (concat str "%s  "))
        (org-timer-set-timer "0:05:00")))

    (message "%s" (concat str "started"))
    (setq org-timer-done-hook nil)
    (add-hook 'org-timer-done-hook (lambda ()
                                     (message "%s" (concat str "ended"))
                                     (ding)))))

(zf/leader
  "op" '(:ignore t :wk "pomodoro")
  "opw" '(zf/start-pomo-work :wk "start work timer")
  "opp" '(org-timer-pause-or-continue :wk "pause/continue")
  "ops" '(org-timer-stop :wk "stop")
  "opb" '(zf/start-pomo-break :wk "start break timer"))
